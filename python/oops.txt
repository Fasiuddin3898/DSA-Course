**Private(like) attributes and methods:Private attributes and classes are meant to be used only within the class and are not
accessible from outside the class
we can define private attribute by putting __ in front of attribute

#Private attribute and method code
class Account:
    def __init__(self,acc_no,acc_pass):
        self.acc_no=acc_no
        self.__acc_pass=acc_pass

    def reset_password(self):
        print(f'password {self.__acc_pass}')

    def __hello(self):  #can not be accessed outside the class
        print(f'welcome')

    def welcome(self):
        self.__hello()

acc1=Account(12345,"abcde")
acc1.reset_password() #this will print the password
acc1.welcome() # this will print welcome
acc1.__hello() # this will throw an error
print(acc1.__acc_pass) #this will throw the error

**Inheritence:When one class(child/derived) derives the properties or methods of another class(parent/base)
#Inheritance
class Car:
    @staticmethod
    def start():
        print(f'car started..')

    @staticmethod
    def stop():
        print(f'car stopped')

class ToyotaCar(Car):
    def __init__(self,name):
        self.name=name

car1=ToyotaCar("fortuner")
car2=ToyotaCar("prius")
car1.start()
car2.stop()
#output
car started..
car stopped

**Super Method:super() method is used to access the methods of the parent class

**Property:We use @Property decorator on any method in the class to use the method as a property.
class Student:
    def __init__(self,phy,scie,math):
        self.phy=phy
        self.scie=scie
        self.math=math
        self.percentage=str((self.phy+self.scie+self.math)/3)+"%"
    @property
    def percentage1(self):
        return str((self.phy+self.scie+self.math)/3)+"%"


stu1=Student(99,98,97)
print(stu1.percentage)
stu1.math=100
print(stu1.math)
print(stu1.percentage). #98
print(stu1.percentage1). #99

#Polymorphism using dunder functions
class Complex:
    def __init__(self,real,img):
        self.real=real
        self.img=img

    def showNumber(self):
        print(self.real,"i +",self.img,"j")

    def addNumber(self,num2):
        new_real=self.real+num2.real
        new_real=self.img+num2.img
        return Complex(new_real,new_real)
    
    def __add__(self,num2):
        new_real=self.real+num2.real
        new_real=self.img+num2.img
        return Complex(new_real,new_real)


num1=Complex(1,3)
num1.showNumber()

num2=Complex(4,5)
num2.showNumber()

num3=num1.addNumber(num2)
num3.showNumber()

num4=num1+num2
num4.showNumber()
